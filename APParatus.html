Welcome to APParatus

APParatus is a tool designed to help hobbyists with troubleshooting electronic circuits and microcontrollers. The easy-to-use interface allows you to control each individual GPIO on a microcontroller wirelessly and in real time. This helps to cut down on the frustration of trying to figure out why something isn't working correctly.

How to Get Started

First, go to the "Control" tab and select a board. The board name should be printed on your microcontroller, but if you are unsure or your board is not listed, enter the number of GPIO pins. If you do enter the number GPIOs, you will need to create your own code.

Second, go to the "Code" tab and input the required information. Once you do, you will need to upload the given code to your microcontroller. Recommended platforms to do this are PlatformIO, Arduino IDE, Espressif IDE, or PrograminoIDE.

Third, once the code is uploaded, you will see the IP address printed in the serial monitor. Go back to the "Control" tab and input the IP address given to you. Once you do, you will be able to control your board's GPIO pins.

#include <.h>
#include <WebSocketsServer.h>

const char* ssid = "";
const char* password = "";

const int usablePins[] = ;
const int totalPins = sizeof(usablePins) / sizeof(usablePins[0]);

WebSocketsServer webSocket = WebSocketsServer(81);

void webSocketEvent(uint8_t num, WStype_t type, uint8_t * payload, size_t length) {
  if (type == WStype_TEXT) {
    char buffer[length + 1];
    memcpy(buffer, payload, length);
    buffer[length] = '\0';

    char* token = strtok(buffer, ",");
    int pinIndex = 0;

    // writes the 9 pins listed above
    while (token != NULL && pinIndex < totalPins) {
      int GPIOvalue = atoi(token);
      
      analogWrite(usablePins[pinIndex], GPIOvalue); 

      pinIndex++;
      token = strtok(NULL, ",");
    }
  }
}

void setup() {
  Serial.begin(115200);

  delay(500);

  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) { 
  Serial.print(".");
  delay(500); 
  }

  Serial.println("Connected to WiFi!");

  Serial.println("-------------------------------------------");
  Serial.print("This is your IP address, copy and paste it: ");
  Serial.println(WiFi.localIP());
  Serial.println("-------------------------------------------");

  delay(30);
  Serial.end();
  delay(30);

  webSocket.begin();
  webSocket.onEvent(webSocketEvent);

  for (int i = 0; i < totalPins; i++) {
  pinMode(usablePins[i], OUTPUT);
  delay(1);
  }
}

void loop() {
  webSocket.loop();
}
